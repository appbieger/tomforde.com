{%- comment -%} Include: gallery-from-dir.html Usage: {% include gallery-from-dir.html dir='images/stella' gallery_id='stella' %} Params: - dir Ordnerpfad relativ zum Site-Root (OHNE führendes /, z. B. 'images/stella') - gallery_id (optional): ID/Name für data-pswp-gallery (Default: letzter Ordnername) Hinweis: - Bilder werden alphabetisch nach Dateiname sortiert - Unterstützte Formate: .jpg, .jpeg, .png, .webp - Dimensionen werden aus Dateinamen extrahiert (Format: dateiname_BREITExHÖHE.ext) - Beschreibungen werden aus _data/gallery_captions.yml gelesen - Führe 'npm run images:add-dimensions' aus, um Dimensionen zu Dateinamen hinzuzufügen {%- endcomment -%} {%- assign gallery_dir = include.dir -%} {%- assign gallery_id = include.gallery_id | default: 'gallery' -%} {%- comment -%} Filtere alle statischen Dateien nach dem Verzeichnis {%- endcomment -%} {%- assign gallery_images = site.static_files | where_exp: "file", "file.path contains gallery_dir" -%} {%- comment -%} Lade Captions aus Data-Datei {%- endcomment -%} {%- assign captions = site.data.gallery_captions[gallery_id] -%}
{%- for image in gallery_images -%} {%- assign extension = image.extname | downcase -%} {%- comment -%} Ignoriere WebP-Versionen - nur Originale anzeigen {%- endcomment -%} {%- if extension == '.jpg' or extension == '.jpeg' or extension == '.png' -%} {%- comment -%} Extrahiere Dimensionen aus Dateiname (Format: name_WIDTHxHEIGHT.ext) {%- endcomment -%} {%- assign basename = image.basename -%} {%- assign parts = basename | split: '_' -%} {%- assign last_part = parts | last -%} {%- comment -%} Prüfe ob letzter Teil Dimensionen enthält (z.B. "1920x1080") {%- endcomment -%} {%- if last_part contains 'x' -%} {%- assign dimensions = last_part | split: 'x' -%} {%- assign width = dimensions[0] -%} {%- assign height = dimensions[1] -%} {%- comment -%} Entferne Dimensions-Teil für Base-Name {%- endcomment -%} {%- assign name_parts = parts | pop -%} {%- assign base_name = name_parts | join: '_' -%} {%- else -%} {%- comment -%} Keine Dimensionen im Dateinamen, verwende Defaults {%- endcomment -%} {%- assign width = 800 -%} {%- assign height = 600 -%} {%- assign base_name = basename -%} {%- endif -%} {%- comment -%} Hole Beschreibung aus Data-Datei {%- endcomment -%} {%- assign description = captions[base_name] -%} {%- if description and description != '' -%} {{ description }} {%- endif -%} {%- endif -%} {%- endfor -%}
